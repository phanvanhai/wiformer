# Train (wiformer_v4, wiformer): Adam, lr đỉnh 1e-3, warmup 5 epoch + cosine về 0 ở epoch 100,
#   KHÔNG dừng sớm (patience = 0), checkpoint = epoch có val acc cao nhất.
# Thời gian trên SSHAR-ESP (T4), 100 epoch: v4 full ~3.8 h, wo_fft ~3.3 h, wo_block3 ~3.5 h,
#   wo_block2 ~3.1 h, wo_conv / wo_block1 ~1.3-1.4 h, freq_mean ~0.7 h, hybrid ~3.3 h
#   -> 1 phiên 12 h chạy được ~2-3 biến thể nặng. UT_HAR: ~45 phút / biến thể.

# ---------------------------- CÁC TUỲ CHỌN ----------------------------
# VARIANTS (wiformer_v4, xem: python ablation.py --model=wiformer_v4 --list)
#   Kiến trúc: WiFormer_v4_Architecture.md, WiFormer_Paper.md
#   full       v4 đầy đủ                                                [mô hình đề xuất]
#   wo_fft     token chỉ có Conv (bỏ góc nhìn phổ)
#   wo_conv    token chỉ có phổ (bỏ Conv)
#   wo_block1  Khối 1 = Linear(W, D) trên đoạn thô
#   wo_block2  Khối 2 = trung bình các token Rx x Sub
#   wo_block3  Khối 3 = trung bình qua các window
#   freq_mean  chỉ phổ + trung bình token Rx x Sub (= phổ trung bình qua sub, rx); so với wo_conv
#              -> attention chọn phổ từng sub hay lấy trung bình phổ tốt hơn (~40-50 phút)
#   hybrid     mỗi window 2 nhánh: token Conv -> Khối 2 (attention) và phổ trung bình qua (rx, sub);
#              gộp Linear(2D -> D) theo window rồi vào Khối 3 (~3.3 h)
#   hybrid_drop    hybrid + modality dropout: khi train zero nhánh Khối 2 (h_k) với p = 0.5
#                  (theo mẫu) -> buộc Khối 3 đọc cả phổ trung bình (~3.3 h)
#   freq_mean_e200 freq_mean train 200 epoch (cosine về 0 ở epoch 200), ~1.4 h.
#                  EXTRA KHÔNG được đặt --epochs (EXTRA ghi đè tham số của biến thể)
#   nonorm     full, không chuẩn hoá theo mẫu trong mô hình -> dùng cho UT_HAR
#
# VARIANTS (wiformer v2, MODEL = 'wiformer'; python ablation.py --list)
#   v2 trên nhánh này dùng CHUNG chuẩn hoá với v4 (trừ mean theo (rx, sub), chia 1 std mỗi rx,
#   không z-score toàn cục) và cùng công thức huấn luyện (patience 0) -> v2 vs v4 chỉ khác Khối 2-3.
#   (v2 gốc với z-score từng (rx, sub) nằm ở nhánh khác)
#   core  : full wo_fft wo_conv wo_block1 wo_block2 wo_block3
#   design: k2_per_rx k2_pe_none k2_pe_rx_sub k3_learned k3_nopos k2_l1 k3_l2 no_norm all_ant win32 win128
#   window: b_nowin b_nooverlap
#
# EXTRA (tham số thêm cho main.py, áp dụng cho mọi biến thể trong VARIANTS)
#   Huấn luyện
#     --epochs=100            số epoch (mặc định 100; vd --epochs=2 để chạy thử)
#     --patience=0            dừng sớm sau N epoch val không tăng (mặc định 0 = tắt)
#     --lr=1e-3  --optimizer=adam|adamw  --weight_decay=0  --batch_size=32  --warmup_epochs=5
#     --class_weights=auto    CE có trọng số: 'auto' (nghịch tần suất tập train) hoặc
#                             danh sách K số, vd '--class_weights=1,1,0.8,1,0.8,1.5,2'
#     --scheduler=reducelronplateau --scheduler-patience=4 --scheduler-gamma=0.5
#                             (thay cosine bằng giảm lr khi val chững lại)
#   Cửa sổ (v4)
#     --window_sec=0.3        độ dài cửa sổ (giây), stride = W/2
#                             W = lũy thừa 2 gần nhất (ESP 64 = 0.32 s, UT_HAR 32 = 0.256 s)
#     --window=64 --stride=32 đặt trực tiếp theo số mẫu (ghi đè window_sec)
#     --fs=200                tần số lấy mẫu (mặc định sshar_esp 200 Hz, ut_har 125 Hz)
#   Kiến trúc (v4)
#     --mod_drop=0.5          hybrid: xác suất zero nhánh h_k khi train (mặc định 0)
#     --norm=sub|none         sub = trừ mean theo (rx, sub), chia std theo rx (mặc định);
#                             none = không chuẩn hoá trong mô hình (biến thể nonorm)
#     --k2_layers=2 --k3_layers=6   số lớp Khối 2 / Khối 3
#     --d_model=64 --k2_heads=4 --k3_heads=4 --dropout=0.1
#     --k1_conv=3 --k1_pool=flat4   số lớp Conv1d / kiểu pool của Khối 1
#
# SPLIT_MODE / CASE
#   Tất cả 6:2:2:
#   repeat              theo lần lặp (mặc định)
#                       sshar: dir00 rep 1-6/7-8/9-10, dir01/02 rep 1-3/4/5 -> 3840/1280/1280
#                       xrf55: rep 1-12/13-16/17-20
#   user                cross-user 5-fold, CASE 0..4. sshar: 6/2/2 người -> 3840/1280/1280
#                       mỗi người được test đúng 1 lần và làm val đúng 1 lần
#                       fold 0: train 1,3,4,9,11,12  | val 2,10  | test 13,14
#                       fold 1: train 1,4,9,12,13,14 | val 3,11  | test 2,10
#                       fold 2: train 2,4,10,12,13,14| val 1,9   | test 3,11
#                       fold 3: train 2,3,10,11,13,14| val 4,12  | test 1,9
#                       fold 4: train 1,2,3,9,10,11  | val 13,14 | test 4,12
#   position            cross-position (sshar), CASE 0..3: test = 2/8 vị trí, còn lại train:val = 3:1
#   (ut_har bỏ qua SPLIT_MODE, luôn dùng split có sẵn; thư mục ghi 'official')
#
# Output (mỗi run một thư mục <dataset>_<split>_<variant>, vd sshar_esp_repeat_full,
#         ut_har_official_nonorm, sshar_esp_user_c0_full):
#   config.json, metrics.json (test), history.csv, learning_curve.png, best_model.pth
#   summary_splits.csv                      acc / P / R / F1 của train, val, test
#   train/ val/ test/                       với mỗi tập:
#       metrics.json, classification_report.csv (theo lớp),
#       confusion_matrix_count.csv (số mẫu), confusion_matrix_rowpct.csv (% theo hàng),
#       confusion_matrix.png (số mẫu + %), predictions.csv (true / pred / xác suất)

# ---- đường dẫn dataset (thư mục Input trên Kaggle) ----

# UT_HAR          : thư mục chứa data/X_train.csv và label/y_train.csv (hoặc thư mục cha)
DATA_PATHS = {
    'sshar_esp': '/kaggle/input/datasets/phanvanhai/cmr-har-room2',
    'ut_har':    '/kaggle/input/datasets/phanvanhai/ut-har/UT_HAR',
}

REPO_URL   = 'https://github.com/phanvanhai/wiformer.git'
BRANCH     = 'cross-user'
MODEL      = 'wiformer_v4'   # wiformer_v4 | wiformer (v2)
DATASET    = 'sshar_esp'     # sshar_esp | ut_har
SPLIT_MODE = 'user'          # repeat | user | position (đều 6:2:2) ; ut_har: bỏ qua
CASE       = 0               # user: fold 0..4 / position: 0..3
VARIANTS   = ['freq_mean', 'freq_mean_e200', 'full']   # danh sách đầy đủ: xem "CÁC TUỲ CHỌN" ở đầu file
EXTRA      = ''              # xem "CÁC TUỲ CHỌN"; mặc định 100 epoch; vd '--epochs=2' để chạy thử
RUN_TAG    = 'ablation_v4'
PREV_RESULTS = ''            # vd '/kaggle/input/<output-phien-truoc>/results/ablation_v4'

import os, subprocess, shutil, shlex, time
from pathlib import Path
CODE_DIR, OUTPUT = '/kaggle/working/cmr_har_code', '/kaggle/working'
DATA_ROOT = DATA_PATHS[DATASET]
assert Path(DATA_ROOT).is_dir(), f'Không thấy {DATA_ROOT}: kiểm tra Add Input / DATA_PATHS'
print('DATA_ROOT =', DATA_ROOT); print(sorted(os.listdir(DATA_ROOT))[:10])

subprocess.run(['rm', '-rf', CODE_DIR], check=True)

# ============================== CELL 2 ==============================
import pandas as pd

def run(cmd):
    print('$', ' '.join(cmd), flush=True)
    proc = subprocess.Popen(cmd, cwd=CODE_DIR, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    for line in proc.stdout:
        print(line, end='', flush=True)
    return proc.wait()

RESULT_BASE = f'{OUTPUT}/results/{RUN_TAG}'
os.makedirs(RESULT_BASE, exist_ok=True)
if PREV_RESULTS and Path(PREV_RESULTS).is_dir():          # run đã xong sẽ bị --skip_done bỏ qua
    shutil.copytree(PREV_RESULTS, RESULT_BASE, dirs_exist_ok=True)

t0 = time.time()
run(['python', '-u', 'ablation.py', f'--model={MODEL}', f'--data_root={DATA_ROOT}',
     f'--dataset={DATASET}', f'--result_root={RESULT_BASE}', '--skip_done',
     f'--split_mode={SPLIT_MODE}', f'--case_split={CASE}',
     '--variants', *VARIANTS, '--extra', *shlex.split(EXTRA)])
print(f'\nTotal time: {(time.time() - t0) / 3600:.2f} h')

# Tổng hợp (kể cả run từ phiên trước): test + train / val, knock-out góc nhìn, hoán vị RX
for csv in sorted(Path(RESULT_BASE).glob('ablation_*.csv')):
    df = pd.read_csv(csv)
    df = df[[c for c in ['folder', 'accuracy', 'f1_macro', 'val_acc', 'train_acc', 'delta_acc',
                         'best_epoch', 'epochs_trained', 'total_params', 'training_time'] if c in df]]
    print('\n#', csv.name); print(df.round(4).to_string(index=False))
for d in sorted(p for p in Path(RESULT_BASE).iterdir() if p.is_dir()):
    print(f'\n######## {d.name}')
    for f in ('summary_splits.csv', 'test/confusion_matrix_rowpct.csv',
              'view_contribution.json', 'rx_permutation.json'):
        if (d / f).exists():
            print(f'# {f}\n' + (d / f).read_text())

zip_path = shutil.make_archive(f'{OUTPUT}/results_{RUN_TAG}', 'zip', f'{OUTPUT}/results', RUN_TAG)
print(zip_path, f'{os.path.getsize(zip_path) / 1e6:.1f} MB')
shutil.rmtree(CODE_DIR, ignore_errors=True)
